1⍝# Minesweeper's rules, a library: minesweeper.xtl (scripted), play.xtl 2⍝# (at the terminal) and the web page all use these. A 9 by 9 field 3⍝# hides 10 mines; open a square to see how many mines touch it, flag 4⍝# the ones you think are mines, and open every safe square to win. 5⍝# 6⍝# The state is one vector: the status (0 playing, 1 won, 2 lost), then 7⍝# the mines, the opened squares and the flags, three 9 by 9 0/1 boards. 8 9ᵇ⁼u̲se< "Board" 10 11⍝ :: Int 12⍝# The field's width and height. 13ˡn ← 9 14⍝ :: a -> a 15⍝# Where the mines are, n by n (1 a mine). 16ˡm̲ines ← { s → (ˡn c̲at ˡn) r̲eshape 81 t̲ake 1 d̲rop s } 17⍝ :: a -> a 18⍝# Which squares are open, n by n. 19ˡo̲pen ← { s → (ˡn c̲at ˡn) r̲eshape 81 t̲ake 82 d̲rop s } 20⍝ :: a -> a 21⍝# Which squares are flagged, n by n. 22ˡf̲lags ← { s → (ˡn c̲at ˡn) r̲eshape -81 t̲ake s } 23⍝ :: a -> a 24⍝# 0 playing, 1 won, 2 lost. 25ˡs̲tatus ← { s → f̲irst s } 26⍝ :: a -> a -> a 27⍝# The state again from a status and the three boards. 28ˡs̲tate ← { st m → st c̲at r̲avel m } 29 30⍝ :: (Any a, Num b, Truthy b) => a -> b 31⍝# A new field: 10 mines on the first 10 squares of a random order. 32ˡn̲ew ← { x → 33 m ← (r̲ange 81) m̲ember? 10 t̲ake g̲rade r̲oll! 81 r̲eshape 1000000 34 0 c̲at m c̲at 162 r̲eshape 0 35} 36 37⍝ :: Num a => a -> a 38⍝# The eight neighbors of every square at once, as Life counts them 39⍝# (lib/Board: the board in a border of zeros rotated by every offset and 40⍝# summed, less itself). 41ˡa̲round ← { b → ᵇa̲round b } 42⍝ :: Num a => a -> a 43⍝# How many mines touch each square. 44ˡc̲ounts ← { s → ˡa̲round ˡm̲ines s } 45 46⍝ :: a -> a -> a 47⍝# The state again from a status and the three boards' parts. 48ˡw̲ith ← { s so → (f̲irst so) c̲at (r̲avel ˡm̲ines s) c̲at 1 d̲rop so } 49 50⍝ :: (Num a, Truthy a) => a -> Int -> a 51⍝# Opening square rc (row, column from 1): a mine loses; otherwise the 52⍝# opened region grows by flood fill: every opened square with no mine 53⍝# around it opens its neighbors, again and again until nothing 54⍝# changes.
